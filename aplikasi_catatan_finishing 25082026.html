Integrasi Google Sheets API

Hubungkan aplikasi FinishLog dengan spreadsheet kantor Anda agar seluruh catatan finishing tersimpan otomatis.

Pastikan URL berakhiran /exec.

Kode Google Apps Script (Code.gs) 
function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Laporan Finishing");
    
    // Auto create sheet tab if missing
    if (!sheet) {
      sheet = ss.insertSheet("Laporan Finishing");
      sheet.appendRow([
        "Timestamp", "Tanggal", "No JO", "No PO", "Nama Konsumen", 
        "Brand", "Bahan / Item", "Ukuran", "QTY", "Mesin", 
        "Operator", "Jenis Finishing", "Catatan", "Shift", 
        "Kepala Finishing", "Kepala Shift"
      ]);
      sheet.getRange(1, 1, 1, 16).setFontWeight("bold").setBackground("#e2e8f0");
    }
    
    var data = JSON.parse(e.postData.contents);
    var timestamp = new Date();
    
    // Check if items array exists (multi-item support)
    if (data.items && data.items.length > 0) {
      for (var i = 0; i < data.items.length; i++) {
        var item = data.items[i];
        sheet.appendRow([
          timestamp,
          data.tanggal || "",
          data.noJO || "",
          data.noPO || "",
          data.konsumen || "",
          data.brand || "",
          item.bahan || "",
          item.ukuran || "",
          item.qty || 0,
          item.mesin || "",
          Array.isArray(data.operator) ? data.operator.join(", ") : (data.operator || ""),
          data.finishing || "",
          data.catatan || "",
          data.shift || "",
          data.kepalaFinishing || "",
          data.kepalaShift || ""
        ]);
      }
    } else {
      sheet.appendRow([
        timestamp,
        data.tanggal || "",
        data.noJO || "",
        data.noPO || "",
        data.konsumen || "",
        data.brand || "",
        data.bahan || "",
        data.ukuran || "",
        data.qty || 0,
        data.mesin || "",
        Array.isArray(data.operator) ? data.operator.join(", ") : (data.operator || ""),
        data.finishing || "",
        data.catatan || "",
        data.shift || "",
        data.kepalaFinishing || "",
        data.kepalaShift || ""
      ]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ "status": "success", "message": "Data berhasil disimpan" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Laporan Finishing");
    if (!sheet) {
      return ContentService.createTextOutput(JSON.stringify({ status: "success", data: [] }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    var rows = sheet.getDataRange().getValues();
    if (rows.length <= 1) {
      return ContentService.createTextOutput(JSON.stringify({ status: "success", data: [] }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    var recordsMap = {};
    for (var i = 1; i < rows.length; i++) {
      var r = rows[i];
      if (!r[2]) continue;
      
      var noJO = String(r[2]).trim();
      var itemObj = {
        bahan: String(r[6] || ""),
        ukuran: String(r[7] || ""),
        qty: Number(r[8]) || 0,
        mesin: String(r[9] || "")
      };
      
      if (!recordsMap[noJO]) {
        var opArr = String(r[10] || "").split(",").map(function(s){ return s.trim(); }).filter(Boolean);
        var tgl = r[1] instanceof Date ? Utilities.formatDate(r[1], Session.getScriptTimeZone(), "yyyy-MM-dd") : String(r[1] || "");
        recordsMap[noJO] = {
          id: "gs_" + noJO,
          tanggal: tgl,
          noJO: noJO,
          noPO: String(r[3] || ""),
          konsumen: String(r[4] || ""),
          brand: String(r[5] || ""),
          items: [itemObj],
          operator: opArr,
          finishing: String(r[11] || ""),
          catatan: String(r[12] || ""),
          shift: String(r[13] || ""),
          kepalaFinishing: String(r[14] || ""),
          kepalaShift: String(r[15] || ""),
          timestamp: r[0] ? new Date(r[0]).toISOString() : new Date().toISOString()
        };
      } else {
        recordsMap[noJO].items.push(itemObj);
      }
    }
    
    var resultList = [];
    for (var k in recordsMap) {
      resultList.push(recordsMap[k]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", data: resultList }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
